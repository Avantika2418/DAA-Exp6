import random
import matplotlib.pyplot as plt


# ---------------- NAIVE STRING MATCHING ----------------
def naive_search(text, pattern):
    n, m = len(text), len(pattern)
    matches = []
    comparisons = 0

    if m == 0 or m > n:
        return matches, comparisons

    for i in range(n - m + 1):
        j = 0

        while j < m:
            comparisons += 1

            if text[i + j] != pattern[j]:
                break

            j += 1

        if j == m:
            matches.append(i)

    return matches, comparisons


# ---------------- KMP ALGORITHM ----------------
def compute_lps(pattern):
    m = len(pattern)
    lps = [0] * m

    length = 0
    i = 1

    while i < m:

        if pattern[i] == pattern[length]:
            length += 1
            lps[i] = length
            i += 1

        elif length != 0:
            length = lps[length - 1]

        else:
            lps[i] = 0
            i += 1

    return lps


def kmp_search(text, pattern):
    n, m = len(text), len(pattern)

    matches = []
    comparisons = 0

    if m == 0 or m > n:
        return matches, comparisons

    lps = compute_lps(pattern)

    i = 0
    j = 0

    while i < n:
        comparisons += 1

        if text[i] == pattern[j]:
            i += 1
            j += 1

            if j == m:
                matches.append(i - j)
                j = lps[j - 1]

        else:
            if j != 0:
                j = lps[j - 1]
            else:
                i += 1

    return matches, comparisons


# ---------------- RABIN-KARP ALGORITHM ----------------
def rabin_karp(text, pattern, q=101):
    n, m = len(text), len(pattern)

    d = 256
    matches = []
    comparisons = 0

    if m == 0 or m > n:
        return matches, comparisons

    h = pow(d, m - 1, q)

    p_hash = 0
    t_hash = 0

    # Calculate initial hash values
    for i in range(m):
        p_hash = (d * p_hash + ord(pattern[i])) % q
        t_hash = (d * t_hash + ord(text[i])) % q

    # Check each window
    for s in range(n - m + 1):

        # Hash values match
        if p_hash == t_hash:

            match = True

            for k in range(m):
                comparisons += 1

                if text[s + k] != pattern[k]:
                    match = False
                    break

            if match:
                matches.append(s)

        # Calculate next window hash
        if s < n - m:
            t_hash = (
                d * (t_hash - ord(text[s]) * h)
                + ord(text[s + m])
            ) % q

    return matches, comparisons


# ---------------- MAIN EXECUTION ----------------

# Example input
text = "AABAACAADAABAABA"
pattern = "AABA"

print("STRING MATCHING ALGORITHMS")
print("=" * 50)

print(f"Text    : {text}")
print(f"Pattern : {pattern}")

m1, c1 = naive_search(text, pattern)
m2, c2 = kmp_search(text, pattern)
m3, c3 = rabin_karp(text, pattern)

print("\nResults:")
print(f"Naive Search  -> Matches: {m1}, Comparisons: {c1}")
print(f"KMP Search    -> Matches: {m2}, Comparisons: {c2}")
print(f"Rabin-Karp    -> Matches: {m3}, Comparisons: {c3}")


# ---------------- PERFORMANCE COMPARISON ----------------

print("\nPERFORMANCE COMPARISON")
print("-" * 50)

# Generate large random text
text_large = "".join(random.choices("ABCD", k=10000))

patterns = [
    "AB",
    "ABCD",
    "ABCDAB",
    "ABCDABCD"
]

naive_comparisons = []
kmp_comparisons = []
rk_comparisons = []

print(f'{"Pattern":>12} {"Naive":>12} {"KMP":>12} {"RK":>12}')
print("-" * 52)

for p in patterns:

    _, c1 = naive_search(text_large, p)
    _, c2 = kmp_search(text_large, p)
    _, c3 = rabin_karp(text_large, p)

    naive_comparisons.append(c1)
    kmp_comparisons.append(c2)
    rk_comparisons.append(c3)

    print(f"{p:>12} {c1:>12} {c2:>12} {c3:>12}")


# ---------------- GRAPH ----------------

x = list(range(len(patterns)))
width = 0.25

plt.figure(figsize=(10, 6))

plt.bar(
    [i - width for i in x],
    naive_comparisons,
    width=width,
    label="Naive"
)

plt.bar(
    x,
    kmp_comparisons,
    width=width,
    label="KMP"
)

plt.bar(
    [i + width for i in x],
    rk_comparisons,
    width=width,
    label="Rabin-Karp"
)

plt.xlabel("Pattern")
plt.ylabel("Number of Character Comparisons")
plt.title("Comparative Analysis of String Matching Algorithms")

plt.xticks(x, patterns)

plt.legend()
plt.grid(axis="y", linestyle="--", alpha=0.5)

plt.tight_layout()
plt.show()
